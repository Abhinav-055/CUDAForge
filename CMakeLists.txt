cmake_minimum_required(VERSION 3.20)
project(CUDAForge VERSION 1.0.0 LANGUAGES CXX)

option(ENABLE_CUDA "Build real CUDA kernels (requires NVIDIA CUDA Toolkit)" OFF)
option(ENABLE_OPENMP "Enable OpenMP CPU implementations when available" ON)
option(BUILD_TESTING "Build correctness tests" ON)
if(BUILD_TESTING)
  include(CTest)
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(ENABLE_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(NOT CMAKE_CUDA_COMPILER)
    message(FATAL_ERROR "ENABLE_CUDA=ON, but nvcc was not found. Install the NVIDIA CUDA Toolkit on a CUDA-capable Linux machine, then configure with -DENABLE_CUDA=ON.")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(CMAKE_CUDA_STANDARD 20)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
else()
  message(STATUS "CUDA disabled: configuring the CPU backend only. For remote NVIDIA GPU builds, install CUDA Toolkit and use -DENABLE_CUDA=ON.")
endif()

add_library(cudaforge_cpu src/cpu/kernels.cpp)
target_include_directories(cudaforge_cpu PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(ENABLE_OPENMP)
  find_package(OpenMP QUIET)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(cudaforge_cpu PUBLIC OpenMP::OpenMP_CXX)
    target_compile_definitions(cudaforge_cpu PUBLIC CUDAFORGE_HAS_OPENMP=1)
    message(STATUS "OpenMP enabled")
  else()
    message(STATUS "OpenMP unavailable; CPU parallel entry points use serial fallback")
  endif()
endif()

if(ENABLE_CUDA)
  add_library(cudaforge_cuda src/cuda/kernels.cu)
  target_include_directories(cudaforge_cuda PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_link_libraries(cudaforge_cuda PUBLIC cudaforge_cpu)
  target_link_libraries(cudaforge_cuda PUBLIC CUDA::cudart)
  target_compile_definitions(cudaforge_cuda PUBLIC CUDAFORGE_ENABLE_CUDA=1)
  set_target_properties(cudaforge_cuda PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
  if(BUILD_TESTING)
    add_executable(cudaforge_cuda_tests tests/test_cuda.cpp)
    target_link_libraries(cudaforge_cuda_tests PRIVATE cudaforge_cuda)
    add_test(NAME cuda_correctness COMMAND cudaforge_cuda_tests)
    set_tests_properties(cuda_correctness PROPERTIES SKIP_RETURN_CODE 77)
  endif()
  add_executable(cudaforge_cuda_bench benchmarks/cuda_benchmark.cpp)
  target_link_libraries(cudaforge_cuda_bench PRIVATE cudaforge_cuda)
endif()

add_executable(cudaforge_bench benchmarks/benchmark.cpp)
target_link_libraries(cudaforge_bench PRIVATE cudaforge_cpu)
if(ENABLE_CUDA)
  target_link_libraries(cudaforge_bench PRIVATE cudaforge_cuda)
  target_compile_definitions(cudaforge_bench PRIVATE CUDAFORGE_ENABLE_CUDA=1)
endif()

if(BUILD_TESTING)
  add_executable(cudaforge_tests tests/test_cpu.cpp)
  target_link_libraries(cudaforge_tests PRIVATE cudaforge_cpu)
  add_test(NAME cpu_correctness COMMAND cudaforge_tests)
endif()
